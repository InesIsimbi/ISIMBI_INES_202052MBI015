year_cols <- paste0(“X”, 1990:2019)

Fill only short INTERNAL gaps (<= 2 consecutive NAs) by linear interpolation

fill_short_gaps <- function(v, max_gap = 2) { n <- length(v); i <- 1 while (i <= n) { if (is.na(v[i])) { j <- i while (j <= n && is.na(v[j])) j <- j + 1 if (i > 1 && j <= n && (j - i) <= max_gap) { v[i:(j - 1)] <- approx(c(i - 1, j), c(v[i - 1], v[j]), xout = i:(j - 1))$y } i <- j } else i <- i + 1 } v } co2_i <- co2 co2_i[year_cols] <- t(apply(co2[year_cols], 1, fill_short_gaps))

———- 3. Merge ———-
